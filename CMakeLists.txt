cmake_minimum_required(VERSION 3.22)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE "Debug")
endif()

set(UGV_NODE_ROLE "LEFT" CACHE STRING "STM32 motor-node role: LEFT or RIGHT")
set_property(CACHE UGV_NODE_ROLE PROPERTY STRINGS LEFT RIGHT)
string(TOUPPER "${UGV_NODE_ROLE}" UGV_NODE_ROLE_UPPER)
string(TOLOWER "${UGV_NODE_ROLE}" UGV_NODE_ROLE_LOWER)

if(NOT UGV_NODE_ROLE_UPPER STREQUAL "LEFT" AND
   NOT UGV_NODE_ROLE_UPPER STREQUAL "RIGHT")
    message(FATAL_ERROR "UGV_NODE_ROLE must be LEFT or RIGHT")
endif()

set(UGV_FIRMWARE_KIND "APPLICATION" CACHE STRING
    "STM32 image kind: APPLICATION or BOOTLOADER")
set_property(CACHE UGV_FIRMWARE_KIND PROPERTY STRINGS APPLICATION BOOTLOADER)
string(TOUPPER "${UGV_FIRMWARE_KIND}" UGV_FIRMWARE_KIND_UPPER)
if(NOT UGV_FIRMWARE_KIND_UPPER STREQUAL "APPLICATION" AND
   NOT UGV_FIRMWARE_KIND_UPPER STREQUAL "BOOTLOADER")
    message(FATAL_ERROR "UGV_FIRMWARE_KIND must be APPLICATION or BOOTLOADER")
endif()

if(UGV_FIRMWARE_KIND_UPPER STREQUAL "BOOTLOADER")
    set(CMAKE_PROJECT_NAME "UGV_BOOTLOADER_${UGV_NODE_ROLE_UPPER}")
else()
    set(CMAKE_PROJECT_NAME "UGV_STM32_${UGV_NODE_ROLE_UPPER}")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

project(${CMAKE_PROJECT_NAME} LANGUAGES C ASM)

message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
message(STATUS "STM32 motor node: ${UGV_NODE_ROLE_UPPER}")
message(STATUS "STM32 firmware kind: ${UGV_FIRMWARE_KIND_UPPER}")

if(UGV_FIRMWARE_KIND_UPPER STREQUAL "BOOTLOADER")
    add_subdirectory(firmware/stm32-bootloader)
else()
    add_subdirectory(firmware/stm32-common)
endif()
